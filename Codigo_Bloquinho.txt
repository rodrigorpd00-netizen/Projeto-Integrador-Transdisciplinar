    private void jButtonAdicionarActionPerformed(java.awt.event.ActionEvent evt) {                                                 
        String produto = jTextFieldProduto.getText();
        String categoria = jComboBoxCategoria.getSelectedItem().toString();
        String quantidade = jTextFieldQuantidade.getText();
        String data = jTextFieldData.getText();
        
        if (produto.isEmpty()|| categoria.isEmpty() || quantidade.isEmpty() || data.isEmpty()){

            JOptionPane.showMessageDialog(this,
                "Por favor, preencha os campos",
                "Tente denovo",
                JOptionPane.ERROR_MESSAGE);
        } else {
            DefaultTableModel modelo = (DefaultTableModel) jTableBloquinho.getModel();
            modelo.addRow(new Object[]{produto,categoria, quantidade, data});

            jTextFieldProduto.setText("");
            jComboBoxCategoria.setSelectedIndex(3);
            jTextFieldQuantidade.setText("");
            //jTextFieldData.setText("");
            

        }        // TODO add your handling code here:
    }

    private void jButtonDeletarActionPerformed(java.awt.event.ActionEvent evt) {

        if (row < 0 ){
            JOptionPane.showMessageDialog(this,
                "Nenhuma linha está selecionada! por favor selecione uma linha",

        } else {
            DefaultTableModel modelo = (DefaultTableModel) jTableBloquinho.getModel();
            modelo.removeRow(row);
        }    // TODO add your handling code here:
    }                                              

    private void jButtonGerarActionPerformed(java.awt.event.ActionEvent evt) {                                             
    // 1. Abre a caixa de diálogo para escolher onde salvar a imagem
    javax.swing.JFileChooser fileChooser = new javax.swing.JFileChooser();
    fileChooser.setDialogTitle("Salvar Exportação como Imagem");
    fileChooser.setFileFilter(new javax.swing.filechooser.FileNameExtensionFilter("Imagem PNG (*.png)", "png"));
    
    int userSelection = fileChooser.showSaveDialog(this);
    
    if (userSelection == javax.swing.JFileChooser.APPROVE_OPTION) {
        java.io.File fileToSave = fileChooser.getSelectedFile();
        
        // Garante a extensão .png no nome do arquivo
        String filePath = fileToSave.getAbsolutePath();
        if (!filePath.toLowerCase().endsWith(".png")) {
            fileToSave = new java.io.File(filePath + ".png");
        }
        
        // 2. IMPORTANTE: Identifique o JScrollPane que envolve a sua tabela.
        // No NetBeans, ao arrastar uma tabela, ele cria automaticamente um contêiner de rolagem.
        // Substitua 'jScrollPane1' pelo nome correto do painel da sua tabela (veja no Navigator).
        javax.swing.JComponent componentToPaint = jScrollPane1; 
        
        // Pega as dimensões reais do componente na tela
        int width = componentToPaint.getWidth();
        int height = componentToPaint.getHeight();
        
        // Se o componente não estiver visível ou renderizado, define um tamanho padrão
        if (width <= 0 || height <= 0) {
            width = 600;
            height = 400;
        }
        
        // 3. Cria a imagem em memória com fundo transparente/RGB
        java.awt.image.BufferedImage image = new java.awt.image.BufferedImage(
                width, height, java.awt.image.BufferedImage.TYPE_INT_RGB);
        
        // Cria o ambiente gráfico para "desenhar" o componente dentro da imagem
        java.awt.Graphics2D g2d = image.createGraphics();
        
        // Ativa suavização para que os textos na imagem fiquem nítidos
        g2d.setRenderingHint(java.awt.RenderingHints.KEY_ANTIALIASING, java.awt.RenderingHints.VALUE_ANTIALIAS_ON);
        g2d.setRenderingHint(java.awt.RenderingHints.KEY_TEXT_ANTIALIASING, java.awt.RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        
        // Faz o componente se desenhar de forma idêntica à tela dentro do arquivo de imagem
        componentToPaint.paint(g2d);
        g2d.dispose(); // Libera os recursos do sistema gráfico
        
        // 4. Grava o arquivo final no disco
        try {
            boolean success = javax.imageio.ImageIO.write(image, "png", fileToSave);
            if (success) {
                javax.swing.JOptionPane.showMessageDialog(this, "Imagem da tabela exportada com sucesso!", "Sucesso", javax.swing.JOptionPane.INFORMATION_MESSAGE);
            } else {
                throw new java.io.IOException("Falha ao renderizar formato PNG.");
            }
        } catch (java.io.IOException e) {
            javax.swing.JOptionPane.showMessageDialog(this, "Erro ao salvar a imagem: " + e.getMessage(), "Erro", javax.swing.JOptionPane.ERROR_MESSAGE);
        }
    }    // TODO add your handling code here:
    }
